\[
\frac{\partial{X'}}{\partial{Y'}} = 
\frac{1}{\sqrt{\hat{b}^{2} - 4\hat{c}(\hat{a}-Y')}}
\]

\[
\frac{\partial{X'}}{\partial{\hat{a}}} =
     \frac{-1}{\sqrt{\hat{b}^{2} - 4\hat{c}(\hat{a}-Y')}}
\]


\[
\frac{\partial{X'}}{\partial{\hat{b}}} =
\frac{-1 + \frac{\hat{b}}{\sqrt{\hat{b}^{2} - 4\hat{c}(\hat{a}-Y')}}}{2\hat{c}}
\]


\[
\frac{\partial{X'}}{\partial{\hat{c}}} =
\frac{-\hat{a} + Y'}{\hat{c}\sqrt{\hat{b}^{2} - 4\hat{c}(\hat{a}-Y')}} - 
\frac{-\hat{b} + \sqrt{\hat{b}^{2} - 4\hat{c}(\hat{a}-Y')}}{2\hat{c}^{2}}
\]
