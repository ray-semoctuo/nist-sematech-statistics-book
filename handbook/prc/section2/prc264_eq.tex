\begin{eqnarray*}
\gamma = 1 - Np^{N-1} + (N-1)p^{N}
\end{eqnarray*}

\vspace*{.25in}

\begin{eqnarray*}
N \cong \frac{1}{4} \frac{(1+p)}{(1-p)} \, \chi^2_{\gamma, 4} +\frac{1}{2}
\end{eqnarray*}

\vspace*{.25in}

\begin{eqnarray*}
\hspace*{-1.25in}
N \cong \frac{1}{4} \frac{(1+0.90)}{(1-0.90)} \, \chi^2_{0.95, 4} +\frac{1}{2} = 0.25(19)(9.488)+0.5 = 45.57 = 46
\end{eqnarray*}
